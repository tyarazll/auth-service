        let isLogin = true;

        function toggleMode() {

        function showAlert(msg, isSuccess) {
            const alertBox = document.getElementById('alertBox');
            alertBox.innerText = msg;
            alertBox.className = `alert ${isSuccess ? 'alert-success' : 'alert-danger'} text-center fw-bold`;
            alertBox.classList.remove('d-none');
        }

        async function handleAuth() {
            const user = document.getElementById('username').value;
            const pass = document.getElementById('password').value;
            const endpoint = isLogin ? '/api/auth/login' : '/api/auth/register';

            if (!user || !pass) {
                showAlert("Username dan password wajib diisi!", false);
                return;
            }

            try {
                // Inilah bagian yang menghubungkan tampilan web ke sistem Node.js kalian!
                const response = await fetch(endpoint, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ username: user, password: pass })
                });

                const data = await response.json();

                if (response.ok) {
                    if (isLogin) {
                        document.getElementById('authCard').style.display = 'none';
                        document.getElementById('dashboard').style.display = 'block';
                        document.getElementById('tokenDisplay').innerText = data.token;
                    } else {
                        showAlert("Registrasi sukses! Silakan login.", true);
                        setTimeout(toggleMode, 1500); // Otomatis pindah ke form login
                    }
                } else {
                    // Kalau kena Rate Limit (429) atau password salah, akan muncul di sini
                    showAlert(data.error || data.message || "Terjadi kesalahan!", false);
                }
            } catch (err) {
                showAlert("Gagal terhubung ke server Node.js", false);
            }
        }

        function logout() {
            document.getElementById('username').value = '';
            document.getElementById('password').value = '';
            document.getElementById('dashboard').style.display = 'none';
            document.getElementById('authCard').style.display = 'block';
            document.getElementById('alertBox').classList.add('d-none');
        }
    </script>
